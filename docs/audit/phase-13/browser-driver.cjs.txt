// Uses installed Chrome/CDP and a real production standalone build; no dependencies or fabricated weather.
const fs = require('node:fs');
const path = require('node:path');
const { spawn } = require('node:child_process');
const assert = require('node:assert/strict');

const dir = path.resolve('docs/audit/phase-13/evidence');

fs.cpSync('public', '.next/standalone/public', { recursive: true });

const server = spawn(process.execPath, ['--use-system-ca', '.next/standalone/server.js'], {
  env: { ...process.env, PORT: '3192', HOSTNAME: '0.0.0.0' }, windowsHide: true, stdio: ['ignore', output, output],
});
const chrome = spawn('C:/Program Files/Google/Chrome/Application/chrome.exe', [
  '--headless=new', '--remote-debugging-port=9342', '--remote-debugging-address=127.0.0.1',
  `--user-data-dir=${path.resolve('.next/phase13-chrome-' + mode)}`, '--no-first-run', '--no-default-browser-check', 'about:blank',

const report = { mode, node: process.version, started: new Date().toISOString(), checks: [], events: [], requests: [], screenshots: [] };
let socket, next = 0;
const pending = new Map();
const pause = ms => new Promise(resolve => setTimeout(resolve, ms));
const save = () => fs.writeFileSync(path.join(dir, `${mode}-browser.json`), JSON.stringify(report, null, 2) + '\n');
async function send(method, params = {}) {
  const id = ++next;
  const result = new Promise((resolve, reject) => {
    pending.set(id, { resolve, reject });
    setTimeout(() => { if (pending.delete(id)) reject(new Error('Timeout ' + method)); }, 30000).unref();
  });
  socket.send(JSON.stringify({ id, method, params }));
  return result;
}
async function evaluate(expression) {
  const r = await send('Runtime.evaluate', { expression, awaitPromise: true, returnByValue: true });
  if (r.exceptionDetails) throw new Error(JSON.stringify(r.exceptionDetails));
  return r.result.value;
}
async function waitFor(expression, timeout = 45000) {
  const end = Date.now() + timeout;
  while (Date.now() < end) { if (await evaluate(expression)) return; await pause(150); }
  throw new Error('Condition: ' + expression);
}
async function key(code, type = 'keyDown', extra = {}) {
  await send('Input.dispatchKeyEvent', { type, code, key: code.startsWith('Key') ? code.slice(3).toLowerCase() : code, ...extra });
}
async function tap(code) { await key(code); await key(code, 'keyUp'); }
async function focus() { await evaluate("document.querySelector('.world-canvas').focus()"); }
async function snapshot() {
  return evaluate("({...document.querySelector('.world-root').dataset,geo:document.querySelector('[data-testid=geo-telemetry]')?.innerText,map:document.querySelector('.world-map-panel')?.innerText,debug:document.querySelector('.world-debug')?.innerText})");
}
async function shot(name) {
  const image = await send('Page.captureScreenshot', { format: 'png' });
  fs.writeFileSync(path.join(dir, `${mode}-${name}.png`), Buffer.from(image.data, 'base64'));
  report.screenshots.push(`${mode}-${name}.png`);
}
async function click(selector) {
  const point = await evaluate(`(()=>{const r=document.querySelector(${JSON.stringify(selector)}).getBoundingClientRect();return{x:r.x+r.width/2,y:r.y+r.height/2}})()`);
  await send('Input.dispatchMouseEvent', { type: 'mousePressed', button: 'left', clickCount: 1, ...point });
  await send('Input.dispatchMouseEvent', { type: 'mouseReleased', button: 'left', clickCount: 1, ...point });
}
async function check(name, fn) {
  if (process.argv[3] && !process.argv[3].split(',').includes(name)) return;
  try { const result = await fn(); report.checks.push({ name, status: 'PASS', result }); }
  catch (e) { report.checks.push({ name, status: 'FAIL', error: String(e.stack) }); }
  save();
  console.log(name + ': ' + report.checks.at(-1).status);
}
async function measure() {
  return evaluate('new Promise(resolve=>{let start=performance.now(),last=start;const a=[];function frame(now){a.push(now-last);last=now;if(now-start<3000)requestAnimationFrame(frame);else{a.sort((x,y)=>x-y);resolve({fps:1000*a.length/(now-start),averageMs:(now-start)/a.length,p95Ms:a[Math.floor(a.length*.95)],frames:a.length})}}requestAnimationFrame(frame)})');
}
async function openDebug() {
  await focus();
  for (let i = 0; i < 3; i++) {
    if (await evaluate("!!document.querySelector('.world-debug.ds-surface-card')")) return;
    await tap('F3'); await pause(200);
  }
  throw new Error('Debug did not expand');
}
async function viewport(width, height, touch = false) {
  await send('Emulation.setDeviceMetricsOverride', { width, height, deviceScaleFactor: 1, mobile: touch });
  await send('Emulation.setTouchEmulationEnabled', { enabled: touch, maxTouchPoints: touch ? 5 : 1 });
}
async function navigate(locale = 'en') {
  await send('Page.navigate', { url: 'http://localhost:3192/' + locale });
  await waitFor("document.querySelector('.world-root')?.dataset.runtimeStatus==='ready'");
  await pause(2000);
}
async function main() {

  await new Promise(resolve => socket.addEventListener('open', resolve, { once: true }));

    const m = JSON.parse(data);
    if (m.method === 'Network.requestWillBeSent') report.requests.push({url:m.params.request.url,method:m.params.request.method});
    if (m.id) { const p = pending.get(m.id); if (p) { pending.delete(m.id); m.error ? p.reject(m.error) : p.resolve(m.result); } }
    else if (['Runtime.exceptionThrown', 'Runtime.consoleAPICalled', 'Log.entryAdded', 'Network.loadingFailed'].includes(m.method)) report.events.push(m);

  await viewport(1920, 1080);

    assert.equal(await evaluate("document.querySelectorAll('canvas').length"), 2);
    await shot('1920x1080');
    return snapshot();
  });

  await openDebug();

    await shot('debug');
    return { checks, snapshot: await snapshot(), performance: await measure() };

  await check('walk-east-north-sprint-jump-reset', async () => {
    await focus(); await tap('KeyR'); await pause(500);
    const start = await snapshot();

    const east = await snapshot(); assert(Number(east.playerX) > Number(start.playerX));
    await key('KeyW'); await key('ShiftLeft'); await pause(1000); await key('Space'); await pause(180);
    const jump = await snapshot();
    assert(Number(jump.playerZ) < Number(east.playerZ)); assert.equal(jump.playerState, 'JUMP');
    await key('Space', 'keyUp'); await key('KeyW', 'keyUp'); await key('ShiftLeft', 'keyUp'); await pause(1200);
    const finish = await snapshot();
    if (mode !== 'baseline') { assert.notEqual(start.geo, east.geo); assert.notEqual(east.geo, finish.geo); }
    await tap('KeyR'); await pause(600);

    return { start, east, jump, finish };
  });
  if (mode === 'baseline') return;

    assert(!report.requests.some(r => /overpass|geofabrik|copernicus|wikimedia|gisportal|vnsdi/.test(r.url)));

  });
  await check('map-follow-zoom-guide', async () => {
    await focus(); await tap('KeyF'); await tap('KeyM'); await tap('KeyH'); await pause(300);

  await check('debug-repeated-toggle', async () => {

    const jump = await rect('.world-touch-jump');
    await send('Input.dispatchTouchEvent', { type: 'touchStart', touchPoints: [{ ...jump, id: 5 }] });

    await send('Input.dispatchTouchEvent', { type: 'touchEnd', touchPoints: [] });

    return { pinch, running, jumping };
  });
  await check('vietnamese-F3', async () => {
    await viewport(390,844,true);
    await navigate('vi'); await openDebug(); await pause(600);
    assert((await snapshot()).geo.includes('CHƯA XÁC MINH ĐỊA LÝ'));
    await evaluate("document.querySelector('[data-testid=source-registry] summary').scrollIntoView({block:'center'})");
    await click('[data-testid=source-registry] summary');
    const sources = await evaluate("document.querySelector('[data-testid=source-registry]').innerText");
    assert(sources.includes('6 nguồn')); assert(sources.includes('Chưa tải hình học địa lý thật'));

  await check('weather-real-failure', async () => {
    report.networkNegativeStart = report.events.length;

      await waitFor(`document.documentElement?.lang==='${locale}' && !!document.querySelector('h1') && !document.querySelector('.world-root')`);

    await send('Page.navigate',{url:'http://localhost:3192/'});

    await check('known-interaction-cooldown', async () => {
      await navigate(); await focus(); await tap('KeyR'); await key('KeyW');

}
main().catch(error => { report.failure = String(error.stack); process.exitCode = 1; }).finally(() => {
  report.finished = new Date().toISOString(); save(); socket?.close(); chrome.kill(); server.kill(); fs.closeSync(output);
  console.log(JSON.stringify({ mode, failures: report.checks.filter(c => c.status === 'FAIL').map(c => c.name), fatal: report.failure }));
});
